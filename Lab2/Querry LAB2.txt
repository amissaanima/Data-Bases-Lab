USE [LABA2]


CREATE TABLE ClientsTable(
	IDClient INT NOT NULL PRIMARY KEY,
	FIO VARCHAR(40) NOT NULL,
	Address VARCHAR(40) NOT NULL,
	Passport VARCHAR(15) NOT NULL,
	PhoneNumber VARCHAR(12) NOT NULL,
	Discount INT
)
--Тут идёт приведение к третьей нф таблицы TourTable
CREATE TABLE Country(
    IDCountry INT NOT NULL PRIMARY KEY,
    CountryName NVARCHAR(61)
)

CREATE TABLE Region(
    IDRegion INT NOT NULL PRIMARY KEY,
    IDCountry INT NOT NULL FOREIGN KEY REFERENCES Country([IDCountry]),
    RegionName NVARCHAR(42)
)
-----------------------------------------------------

CREATE TABLE TourTable(
    IDTour INT NOT NULL PRIMARY KEY,
    Program VARCHAR(300) NOT NULL,
    Transport VARCHAR(30) NOT NULL,
    IDRegion INT NOT NULL FOREIGN KEY REFERENCES Region([IDRegion]),
    Duration INT NOT NULL
)

CREATE TABLE HotelsTable(
    IDHotel INT NOT NULL PRIMARY KEY,
    HotelType INT NOT NULL,
    Location VARCHAR(70),
    HotelName VARCHAR(50) NOT NULL
)

CREATE TABLE TourVariantTable(
    IDTourVar INT NOT NULL PRIMARY KEY,
    IDHotel INT NOT NULL FOREIGN KEY REFERENCES HotelsTable([IDHotel]),
    IDTour INT NOT NULL FOREIGN KEY REFERENCES TourTable([IDTour]),
    Cost INT NOT NULL
)

CREATE TABLE OrdersTable(
	IDOrder INT NOT NULL PRIMARY KEY,
	IDVarTour INT NOT NULL FOREIGN KEY REFERENCES TourVariantTable([IDTourVar]),
	IDClient INT NOT NULL FOREIGN KEY REFERENCES ClientsTable([IDClient]),
	TotalCost MONEY NOT NULL,
	RegistrationDate DATE NOT NULL
)

CREATE TABLE ClaimsTable(
	IDClaim INT NOT NULL PRIMARY KEY,
    IDOrder INT NOT NULL FOREIGN KEY REFERENCES OrdersTable([IDOrder]),
    Reply VARCHAR(20),
    Status VARCHAR(20) NOT NULL,
    Description VARCHAR(300) NOT NULL
)
